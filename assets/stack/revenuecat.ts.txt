import { Hono } from 'hono';
import { z } from 'zod';
import { timingSafeEqual } from 'node:crypto';
import { requireSession } from '../security';
import { billingEnvironment } from '../billing-mode';
import type { AppEnv } from '../environment';
export const revenuecatRoutes = new Hono<AppEnv>();
const hookConfig = z.object({
  REVENUECAT_WEBHOOK_AUTH: z.string().min(1),
  REVENUECAT_APP_ID: z.string().min(1),
});
function authorized(actual: string | undefined, expected: string) {
  const a = new TextEncoder().encode(actual);
  const b = new TextEncoder().encode(expected);
  if (a.byteLength !== b.byteLength) return false;
  return timingSafeEqual(a, b);
}
revenuecatRoutes.post('/webhooks/revenuecat', async (c) => {
  const config = hookConfig.safeParse(c.env);
  if (!config.success) return c.json({ error: 'Webhook needs configuration' }, 503);
  if (!authorized(c.req.header('authorization'), config.data.REVENUECAT_WEBHOOK_AUTH))
    return c.json({ error: 'Unauthorized' }, 401);
  const schema = z.object({
    event: z.object({
      id: z.string().min(1).max(512),
      environment: z.literal({ sandbox: 'SANDBOX', live: 'PRODUCTION' }[billingEnvironment(c.env)]),
      app_id: z.literal(config.data.REVENUECAT_APP_ID),
    }),
  });
  const body = schema.safeParse(await c.req.json().catch(() => null));
  if (!body.success) return c.json({ error: 'Invalid webhook or scope' }, 400);
  await c.env.DB.prepare(
    'INSERT OR IGNORE INTO billing_receipt(provider,event_id,received_at) VALUES (?,?,?)',
  )
    .bind('revenuecat', body.data.event.id, Date.now())
    .run();
  return c.json({ received: true });
});
const statusConfig = z.object({
  REVENUECAT_SECRET_KEY: z.string().min(1),
  REVENUECAT_ENTITLEMENT: z.string().min(1),
});
const subscriberSchema = z.object({
  subscriber: z.object({
    entitlements: z.record(
      z.string(),
      z.object({ expires_date: z.string().nullable(), product_identifier: z.string() }),
    ),
    subscriptions: z.record(z.string(), z.object({ is_sandbox: z.boolean() })),
  }),
});
type Subscriber = z.infer<typeof subscriberSchema>['subscriber'];
const current = (expires: string | null) => expires === null || Date.parse(expires) > Date.now();
function entitled(subscriber: Subscriber, key: string, sandbox: boolean) {
  const item = subscriber.entitlements[key];
  if (!item) return false;
  const product = subscriber.subscriptions[item.product_identifier];
  if (!product) return false;
  return product.is_sandbox === sandbox && current(item.expires_date);
}
export async function revenuecatEntitled(env: AppEnv['Bindings'], userId: string) {
  const config = statusConfig.parse(env);
  const response = await fetch(
    'https://api.revenuecat.com/v1/subscribers/' + encodeURIComponent(userId),
    {
      headers: { Authorization: 'Bearer ' + config.REVENUECAT_SECRET_KEY },
      signal: AbortSignal.timeout(10000),
    },
  );
  if (!response.ok) throw Error('RevenueCat unavailable');
  const data = subscriberSchema.parse(await response.json());
  return entitled(
    data.subscriber,
    config.REVENUECAT_ENTITLEMENT,
    billingEnvironment(env) === 'sandbox',
  );
}
revenuecatRoutes.get('/billing/mobile-status', requireSession, async (c) => {
  try {
    return c.json({
      entitled: await revenuecatEntitled(c.env, c.get('userId')),
      provider: 'revenuecat',
      environment: billingEnvironment(c.env),
    });
  } catch {
    return c.json({ error: 'Billing needs configuration or is temporarily unavailable' }, 503);
  }
});
